SELECT ?medicationStatement ?patient ?rxnormCode ?startDate
WHERE {
  # Search for a FHIR MedicationStatement
  ?medicationStatement a fhir:MedicationStatement ;
                       fhir:subject ?patient ;
                       fhir:medicationCodeableConcept ?rxnormCode ;
                       fhir:effectiveDateTime ?startDate .

  # Filter on RxNorm codes (optional)
  FILTER (datatype(?startDate) = xsd:date)
}
LIMIT 10






PREFIX fhir: <http://hl7.org/fhir/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT ?medicationStatement ?patient ?rxnormCode ?conceptName ?startDate
WHERE {
  # FHIR MedicationStatement from the obda file
  ?medicationStatement a fhir:MedicationStatement ;
                       fhir:subject ?patient ;
                       fhir:medicationCodeableConcept ?rxnormCode ;
                       fhir:effectiveDateTime ?startDate .

  # Join OMOP concept table with drug_concept_id
  ?concept a <http://www.ohdsi.org/omop#Concept> ;
           <http://www.ohdsi.org/omop#conceptId> ?rxnormCode ;
           <http://www.ohdsi.org/omop#conceptName> ?conceptName .

  # Optional: only show RxNorm codes
  FILTER regex(str(?rxnormCode), "^[0-9]+$")
}
LIMIT 20
